if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The ID resolution probe requires Windows and MSVC.")
endif()
add_executable(woby_id_resolution main.cpp)
target_include_directories(woby_id_resolution PRIVATE "${PROJECT_SOURCE_DIR}/src")
target_include_directories(woby_id_resolution SYSTEM PRIVATE "${PROJECT_SOURCE_DIR}/third_party/rapidobj/include")
target_link_libraries(woby_id_resolution PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
woby_enable_project_warnings(woby_id_resolution)
if(MSVC)
    target_compile_options(woby_id_resolution PRIVATE $<$<CONFIG:Release>:/Zi> $<$<CONFIG:Release>:/FAs>)
    target_link_options(woby_id_resolution PRIVATE $<$<CONFIG:Release>:/DEBUG>)
endif()
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME woby_id_resolution_contract COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests.py" "$<TARGET_FILE:woby_id_resolution>")
    set_tests_properties(woby_id_resolution_contract PROPERTIES LABELS "unit;cpu" TIMEOUT 120)
endif()
