# Use the current production tuple-index helper, with checked source anchors.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/obj_mesh.cpp")
file(READ "${PROJECT_SOURCE_DIR}/src/obj_mesh.cpp" obj_source)
string(FIND "${obj_source}" "struct IndexKey {" mapping_start)
string(FIND "${obj_source}" "rapidobj::Result parseObj(" mapping_end)
if(mapping_start LESS 0 OR mapping_end LESS mapping_start)
    message(FATAL_ERROR "OBJ mapping helper source anchors changed")
endif()
math(EXPR mapping_length "${mapping_end} - ${mapping_start}")
string(SUBSTRING "${obj_source}" ${mapping_start} ${mapping_length} mapping_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mapping.inc" "${mapping_source}")
add_executable(woby_simplification_14 main.cpp)
target_include_directories(woby_simplification_14 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(woby_simplification_14 SYSTEM PRIVATE "${PROJECT_SOURCE_DIR}/third_party/rapidobj/include")
target_link_libraries(woby_simplification_14 PRIVATE woby_core nlohmann_json::nlohmann_json)
woby_enable_project_warnings(woby_simplification_14)
if(BUILD_TESTING)
    add_test(NAME woby_simplification_14_contract COMMAND woby_simplification_14 --self-test)
    set_tests_properties(woby_simplification_14_contract PROPERTIES LABELS "unit;cpu")
endif()
