function(dense_set_replace_unique variable anchor replacement)
    string(FIND "${${variable}}" "${anchor}" first)
    if(first EQUAL -1)
        message(FATAL_ERROR "Dense-set extraction: missing anchor ${anchor}")
    endif()
    string(LENGTH "${anchor}" length)
    math(EXPR after "${first}+${length}")
    string(SUBSTRING "${${variable}}" ${after} -1 tail)
    string(FIND "${tail}" "${anchor}" second)
    if(NOT second EQUAL -1)
        message(FATAL_ERROR "Dense-set extraction: duplicate anchor ${anchor}")
    endif()
    string(REPLACE "${anchor}" "${replacement}" replaced "${${variable}}")
    set(${variable} "${replaced}" PARENT_SCOPE)
endfunction()
set(custom_header_input "${PROJECT_SOURCE_DIR}/src/analysis_index.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${custom_header_input}")
file(READ "${custom_header_input}" custom_header)
dense_set_replace_unique(custom_header "#pragma once" "#pragma once\n#include \"index_allocation.h\"")
dense_set_replace_unique(custom_header "struct AnalysisIndex {" "struct TrackedCustomAnalysisIndex {")
dense_set_replace_unique(custom_header "std::vector<std::array<T, N>> keys;"
    "std::vector<std::array<T, N>, DenseSetTrackingAllocator<std::array<T, N>>> keys;")
dense_set_replace_unique(custom_header "std::vector<size_t> buckets;"
    "std::vector<size_t, DenseSetTrackingAllocator<size_t>> buckets;")
string(REGEX MATCHALL "AnalysisIndex<T, N>" custom_type_anchors "${custom_header}")
list(LENGTH custom_type_anchors custom_type_count)
if(NOT custom_type_count EQUAL 2)
    message(FATAL_ERROR "Dense-set extraction: custom function type anchors changed")
endif()
string(REPLACE "AnalysisIndex<T, N>" "TrackedCustomAnalysisIndex<T, N>" custom_header "${custom_header}")
string(REGEX MATCHALL "analysisKeyHash" custom_hash_anchors "${custom_header}")
list(LENGTH custom_hash_anchors custom_hash_count)
if(NOT custom_hash_count EQUAL 4)
    message(FATAL_ERROR "Dense-set extraction: custom hash-name anchors changed")
endif()
string(REPLACE "analysisKeyHash" "trackedCustomKeyHash" custom_header "${custom_header}")
file(CONFIGURE OUTPUT "${dense_set_generated_directory}/tracked_custom_index.h" CONTENT "${custom_header}" @ONLY)

foreach(module IN ITEMS mesh_comparison mesh_duplicates mesh_topology)
    set(input "${PROJECT_SOURCE_DIR}/src/${module}.cpp")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
    file(READ "${input}" contents)
    set(anchor "#include \"analysis_index.h\"")
    string(FIND "${contents}" "${anchor}" first_anchor)
    if(first_anchor EQUAL -1)
        message(FATAL_ERROR "Dense-set extraction: missing include anchor in ${input}")
    endif()
    string(LENGTH "${anchor}" anchor_length)
    math(EXPR after_anchor "${first_anchor}+${anchor_length}")
    string(SUBSTRING "${contents}" ${after_anchor} -1 remainder)
    string(FIND "${remainder}" "${anchor}" second_anchor)
    if(NOT second_anchor EQUAL -1)
        message(FATAL_ERROR "Dense-set extraction: duplicate include anchor in ${input}")
    endif()
    string(REPLACE "${anchor}" "#include \"dense_set_pipeline_index.h\"" dense_contents "${contents}")
    string(REPLACE "${anchor}" "#include \"tracked_custom_pipeline_index.h\"" custom_contents "${contents}")
    file(CONFIGURE OUTPUT "${dense_set_generated_directory}/dense/${module}.cpp" CONTENT "${dense_contents}" @ONLY)
    file(CONFIGURE OUTPUT "${dense_set_generated_directory}/custom/${module}.cpp" CONTENT "${custom_contents}" @ONLY)
    list(APPEND dense_set_sources "${dense_set_generated_directory}/dense/${module}.cpp")
    list(APPEND dense_set_custom_sources "${dense_set_generated_directory}/custom/${module}.cpp")
endforeach()
