# Freeze the pre-adoption baseline so the old measurement and rational oracle
# cannot silently adopt the implementation they are meant to check.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/rational-baseline.cpp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/rational-baseline.cpp" probe_source)
set(adoption_baseline_source "${probe_source}")
foreach(symbol IN ITEMS exactTriangleCollapsed trianglesSelfIntersect intersectionStatus inspectIntersections)
    string(REPLACE "${symbol}" "adoption_${symbol}" adoption_baseline_source "${adoption_baseline_source}")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/production_rational_baseline.cpp" CONTENT "${adoption_baseline_source}" @ONLY)
set(probe_filter_anchor "    if (definitelyDisjoint(aPoints, bPoints, aIds, bIds)) { return false; }")
string(FIND "${probe_source}" "${probe_filter_anchor}" probe_filter_position)
if(probe_filter_position EQUAL -1)
    message(FATAL_ERROR "Item 10 probe expects exactly one disjoint-filter anchor; source changed")
endif()
string(LENGTH "${probe_filter_anchor}" probe_filter_length)
math(EXPR probe_filter_end "${probe_filter_position}+${probe_filter_length}")
string(SUBSTRING "${probe_source}" ${probe_filter_end} -1 probe_remainder)
string(FIND "${probe_remainder}" "${probe_filter_anchor}" probe_second_filter)
if(NOT probe_second_filter EQUAL -1)
    message(FATAL_ERROR "Item 10 probe disjoint-filter anchor occurs more than once; source changed")
endif()
string(FIND "${probe_source}" "const auto a = exact(aPoints), b = exact(bPoints);" probe_rational_anchor)
if(probe_rational_anchor EQUAL -1)
    message(FATAL_ERROR "Item 10 probe rational oracle anchor is missing; source changed")
endif()
foreach(symbol IN ITEMS exactTriangleCollapsed trianglesSelfIntersect intersectionStatus inspectIntersections)
    if(symbol STREQUAL "exactTriangleCollapsed")
        set(replacement probeCollapsed)
    elseif(symbol STREQUAL "trianglesSelfIntersect")
        set(replacement probeBaseline)
    elseif(symbol STREQUAL "intersectionStatus")
        set(replacement probeStatus)
    else()
        set(replacement probeInspect)
    endif()
    string(REPLACE "${symbol}" "${replacement}" probe_source "${probe_source}")
endforeach()
string(REPLACE "#include \"mesh_intersections.h\"" "#include \"probe.h\"" probe_source "${probe_source}")
string(REPLACE
    "    if (definitelyDisjoint(aPoints, bPoints, aIds, bIds)) { return false; }"
    "    ++intersection_probe::counters.calls;\n    if (!intersection_probe::forceRational && definitelyDisjoint(aPoints, bPoints, aIds, bIds)) { ++intersection_probe::counters.filtered; return false; }\n    if (intersection_probe::useCandidate) { return intersection_probe::predicate(aPoints,bPoints,aIds,bIds); }\n    intersection_probe::enteredRational(aPoints,bPoints,aIds,bIds);"
    probe_source "${probe_source}")
string(FIND "${probe_source}" "++intersection_probe::counters.calls;" probe_inserted_counter)
string(FIND "${probe_source}" "intersection_probe::enteredRational(aPoints,bPoints,aIds,bIds);" probe_inserted_oracle)
if(probe_inserted_counter EQUAL -1 OR probe_inserted_oracle EQUAL -1)
    message(FATAL_ERROR "Item 10 probe instrumentation substitution failed")
endif()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/production_instrumented.cpp" CONTENT "${probe_source}" @ONLY)
add_executable(woby_simplification_10_probe main.cpp predicate.cpp "${CMAKE_CURRENT_BINARY_DIR}/production_instrumented.cpp")
add_executable(woby_intersection_adoption_benchmark main.cpp predicate.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/production_instrumented.cpp" "${CMAKE_CURRENT_BINARY_DIR}/production_rational_baseline.cpp")
target_include_directories(woby_intersection_adoption_benchmark PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(woby_intersection_adoption_benchmark PRIVATE woby_core CDT::CDT Boost::multiprecision nlohmann_json::nlohmann_json)
target_compile_definitions(woby_intersection_adoption_benchmark PRIVATE WOBY_PROBE_NORMAL_ALLOCATOR=1)
woby_enable_project_warnings(woby_intersection_adoption_benchmark)
target_include_directories(woby_simplification_10_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(woby_simplification_10_probe PRIVATE woby_core CDT::CDT Boost::multiprecision nlohmann_json::nlohmann_json)
woby_enable_project_warnings(woby_simplification_10_probe)
if(MSVC)
    target_compile_options(woby_simplification_10_probe PRIVATE /WX)
    set_source_files_properties(main.cpp predicate.cpp PROPERTIES COMPILE_OPTIONS /fp:strict)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/production_instrumented.cpp" PROPERTIES COMPILE_OPTIONS /fp:precise)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/production_rational_baseline.cpp" PROPERTIES COMPILE_OPTIONS /fp:precise)
else()
    target_compile_options(woby_simplification_10_probe PRIVATE -Werror)
    set_source_files_properties(main.cpp predicate.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-fno-fast-math")
endif()
