if(NOT WIN32 OR NOT MSVC OR NOT TARGET woby_obj_prototype_benchmark)
    message(FATAL_ERROR "Position mapping requires Windows, MSVC, and WOBY_BUILD_RAPIDOBJ_PROTOTYPE_BENCHMARK.")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(position_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(position_generate_args)
if(TARGET woby_obj_e2e_viewer)
    set(position_traced "${PROJECT_BINARY_DIR}/experiments/obj-profile/e2e-viewer")
    list(APPEND position_generate_args --traced "${position_traced}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${PROJECT_SOURCE_DIR}/src/obj_mesh.cpp"
    "${PROJECT_SOURCE_DIR}/experiments/rapidobj_prototype.cpp"
    "${PROJECT_SOURCE_DIR}/experiments/obj-mapping-reference/obj_mesh.cpp"
    "${PROJECT_SOURCE_DIR}/experiments/obj-mapping-reference/obj_corner_mapping.h"
    "${PROJECT_SOURCE_DIR}/experiments/obj-mapping-reference/benchmark.cpp"
    "${PROJECT_SOURCE_DIR}/experiments/obj-mapping-reference/mapping_reference.h"
    "${PROJECT_SOURCE_DIR}/experiments/obj-mapping-reference/profile_generate.py"
    "${PROJECT_SOURCE_DIR}/experiments/obj-profile/generate.py"
    "${PROJECT_SOURCE_DIR}/experiments/obj-profile/e2e_generate.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
    "${PROJECT_SOURCE_DIR}" "${position_generated}" ${position_generate_args} COMMAND_ERROR_IS_FATAL ANY)

get_target_property(position_sources woby_obj_prototype_benchmark SOURCES)
list(REMOVE_ITEM position_sources "${PROJECT_SOURCE_DIR}/src/mesh_intersection_predicates.cpp")
set(position_cpu_sources)
foreach(source IN LISTS position_sources)
    if(source STREQUAL "src/obj_mesh.cpp")
        list(APPEND position_cpu_sources "${position_generated}/obj_mesh.cpp")
    elseif(source STREQUAL "experiments/rapidobj_prototype.cpp")
        list(APPEND position_cpu_sources "${position_generated}/main.cpp")
    elseif(IS_ABSOLUTE "${source}" OR source MATCHES "^\\$<")
        list(APPEND position_cpu_sources "${source}")
    else()
        list(APPEND position_cpu_sources "${PROJECT_SOURCE_DIR}/${source}")
    endif()
endforeach()
add_executable(woby_position_mapping ${position_cpu_sources})
woby_add_intersection_predicates(woby_position_mapping)
target_include_directories(woby_position_mapping PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/src")
target_include_directories(woby_position_mapping SYSTEM PRIVATE "${PROJECT_SOURCE_DIR}/third_party/rapidobj/include")
target_link_libraries(woby_position_mapping PRIVATE Threads::Threads CDT::CDT Boost::multiprecision nlohmann_json::nlohmann_json psapi)
woby_enable_project_warnings(woby_position_mapping)
list(APPEND woby_runtime_targets woby_position_mapping)

foreach(kind IN ITEMS native traced)
    if(kind STREQUAL "native")
        set(reference woby)
        set(position_obj "${position_generated}/obj_mesh.cpp")
    elseif(TARGET woby_obj_e2e_viewer)
        set(reference woby_obj_e2e_viewer)
        set(position_obj "${position_generated}/traced/obj_mesh.cpp")
    else()
        continue()
    endif()
    get_target_property(viewer_sources ${reference} SOURCES)
    set(position_viewer_sources)
    foreach(source IN LISTS viewer_sources)
        if(source MATCHES "[/\\]obj_mesh\\.cpp$")
            continue()
        elseif(IS_ABSOLUTE "${source}" OR source MATCHES "^\\$<")
            list(APPEND position_viewer_sources "${source}")
        elseif(kind STREQUAL "traced")
            list(APPEND position_viewer_sources "${PROJECT_SOURCE_DIR}/experiments/obj-profile/${source}")
        else()
            list(APPEND position_viewer_sources "${PROJECT_SOURCE_DIR}/${source}")
        endif()
    endforeach()
    set(target "woby_position_mapping_${kind}")
    add_executable(${target} WIN32 ${position_viewer_sources} "${position_obj}")
    foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS LINK_LIBRARIES)
        get_target_property(value ${reference} ${property})
        if(value)
            set_property(TARGET ${target} PROPERTY ${property} "${value}")
        endif()
    endforeach()
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_include_directories(${target} SYSTEM PRIVATE "${PROJECT_SOURCE_DIR}/third_party/rapidobj/include")
    target_compile_definitions(${target} PRIVATE POSITION_MAPPING_FORCE=1)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/Zi>)
    target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/DEBUG>)
    woby_enable_project_warnings(${target})
    woby_enable_enum_switch_warnings(${target})
    list(APPEND woby_runtime_targets ${target})
    set(control "${target}_reference")
    add_executable(${control} WIN32 ${position_viewer_sources} "${position_obj}")
    foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_OPTIONS LINK_OPTIONS LINK_LIBRARIES)
        get_target_property(value ${target} ${property})
        if(value)
            set_property(TARGET ${control} PROPERTY ${property} "${value}")
        endif()
    endforeach()
    get_target_property(control_definitions ${target} COMPILE_DEFINITIONS)
    list(REMOVE_ITEM control_definitions POSITION_MAPPING_FORCE=1)
    target_compile_definitions(${control} PRIVATE ${control_definitions} POSITION_MAPPING_FORCE=0)
    list(APPEND woby_runtime_targets ${control})
endforeach()
set(woby_runtime_targets "${woby_runtime_targets}" PARENT_SCOPE)
if(BUILD_TESTING)
    add_test(NAME woby_position_mapping_contract COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests.py" "$<TARGET_FILE:woby_position_mapping>")
    set_tests_properties(woby_position_mapping_contract PROPERTIES LABELS "unit;cpu" TIMEOUT 120)
endif()
