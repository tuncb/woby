find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(workflow_sources obj_mesh model_mesh background_load ui_state scene_mesh_preparation surface_annotation annotation_preparation main)
set(workflow_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${workflow_generated}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generate.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/graphics.cpp")
foreach(name IN LISTS workflow_sources)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/src/${name}.cpp")
endforeach()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
    "${PROJECT_SOURCE_DIR}" "${workflow_generated}" COMMAND_ERROR_IS_FATAL ANY)
foreach(target IN ITEMS woby woby_core)
    get_target_property(target_sources ${target} SOURCES)
    foreach(name IN LISTS workflow_sources)
        if("src/${name}.cpp" IN_LIST target_sources)
            list(REMOVE_ITEM target_sources "src/${name}.cpp")
            list(APPEND target_sources "${workflow_generated}/${name}.cpp")
        endif()
    endforeach()
    set_property(TARGET ${target} PROPERTY SOURCES "${target_sources}")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endforeach()
target_sources(woby_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/trace.cpp")
get_target_property(graphics_sources woby_graphics SOURCES)
list(REMOVE_ITEM graphics_sources "${PROJECT_SOURCE_DIR}/src/graphics.cpp")
list(APPEND graphics_sources "${workflow_generated}/graphics.cpp")
set_property(TARGET woby_graphics PROPERTY SOURCES "${graphics_sources}")
if(WIN32)
    target_link_libraries(woby_core PRIVATE psapi)
endif()
if(BUILD_TESTING)
    add_test(NAME woby_load_workflow_analysis COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests.py")
endif()
